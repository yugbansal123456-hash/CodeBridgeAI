```cpp
#include <iostream>
using namespace std;

int main() {
    int sum = 0;
    for (int i = 1; i <= 10; i++) {
        sum += i;
    }
    cout << "Sum 1..10 = " << sum << endl;

    int evens = 0;
    for (int i = 1; i <= 20; i++) {
        if (i % 2 == 0) evens += i;
    }
    cout << "Sum of evens up to 20 = " << evens << endl;

    while (n > 1) {
        fact *= n;
        n--;
    }
    cout << "5! = " << fact << endl;

    for (int i = 1; i <= 3; i++) {
        for (int j = 1; j <= 3; j++) {
            cout << i * j;
            if (j < 3) cout << " ";
        }
        cout << endl;
    }

    int countdown = 3;
    do {
        cout << "T-" << countdown << endl;
        countdown--;
    } while (countdown > 0);
    return 0;
```

```cpp
#include <iostream>
using namespace std;

long long factorial(int n) { return n <= 1 ? 1 : n * factorial(n - 1); }
int fib(int n) { return n < 2 ? n : fib(n - 1) + fib(n - 2); }
int gcd(int a, int b) { return b == 0 ? a : gcd(b, a % b); }

long long power(long long base, int exp) {
    if (exp == 0) return 1;
    long long half = power(base, exp / 2);
    return exp % 2 == 0 ? half * half : half * half * base;
}

int main() {
    cout << "10! = " << factorial(10) << endl;
    cout << "15! = " << factorial(15) << endl;
    cout << "Fibonacci:";
    for (int i = 0; i <= 10; i++) cout << " " << fib(i);
    cout << endl;
    cout << "gcd(48, 18) = " << gcd(48, 18) << endl;
    cout << "gcd(17, 5) = " << gcd(17, 5) << endl;
    cout << "2^30 = " << power(2, 30) << endl;
    cout << "3^13 = " << power(3, 13) << endl;
    return 0;
}
```

### T04
```cpp
#include <iostream>
#include <memory>
#include <string>
#include <vector>
using namespace std;

class Shape {
public:
    virtual ~Shape() {}
    virtual double area() const = 0;
    virtual string name() const { return "Shape"; }
};

class Rectangle : public Shape {
protected:
    double w, h;
public:
    Rectangle(double w_, double h_) : w(w_), h(h_) {}
    double area() const override { return w * h; }
    string name() const override { return "Rectangle"; }
};

class Square : public Rectangle {
public:
    explicit Square(double s) : Rectangle(s, s) {}
    string name() const override { return "Square"; }
};

class Circle : public Shape {
    double r;
public:
    explicit Circle(double r_) : r(r_) {}
    double area() const override { return 3.14 * r * r; }
    string name() const override { return "Circle"; }
};

int main() {
    vector<unique_ptr<Shape>> shapes;
    shapes.push_back(make_unique<Rectangle>(3, 4));
    shapes.push_back(make_unique<Square>(5));
    shapes.push_back(make_unique<Circle>(2));

    double total = 0;
    for (const auto& s : shapes) {
        cout << s->name() << " area = " << s->area() << endl;
        total += s->area();
    }
    cout << "Total area = " << total << endl;
    return 0;
}
```

```cpp
#include <iostream>
#include <string>
#include <vector>
using namespace std;

template <typename T>
T biggest(T a, T b) {
    return a > b ? a : b;
}

template <typename T>
T sumAll(const vector<T>& v) {
    T total = T();
    for (const T& x : v) total = total + x;
    return total;
}

template <typename T>
struct Stack {
    vector<T> items;
    void push(T x) { items.push_back(x); }
    T pop() {
        T t = items.back();
        items.pop_back();
        return t;
    }
    bool empty() const { return items.empty(); }
};

int main() {
    cout << biggest(3, 9) << endl;
    cout << biggest(2.5, 1.5) << endl;
    cout << biggest(string("pear"), string("apple")) << endl;

    cout << sumAll(vector<int>{1, 2, 3, 4}) << endl;
    cout << sumAll(vector<double>{1.5, 2.5, 3.0}) << endl;
    cout << sumAll(vector<string>{"a", "b", "c"}) << endl;

    Stack<int> st;
    st.push(10);
    st.push(20);
    st.push(30);
    while (!st.empty()) cout << st.pop() << " ";
    cout << endl;
    return 0;
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

struct Box {
    int v;
};

void byValue(int x) { x += 10; }
void byRef(int& x) { x += 10; }
void byPtr(int* x) { *x += 10; }
void boxByValue(Box b) { b.v = 99; cout << "inside boxByValue: " << b.v << endl; }
void boxByRef(Box& b) { b.v = 99; }
void vecByValue(vector<int> v) { v.push_back(4); }
void vecByRef(vector<int>& v) { v.push_back(4); }

int main() {
    int n = 1;
    byValue(n);
    cout << "after byValue: " << n << endl;
    byRef(n);
    cout << "after byRef: " << n << endl;
    byPtr(&n);
    cout << "after byPtr: " << n << endl;

    Box b{1};
    boxByValue(b);
    cout << "box after boxByValue: " << b.v << endl;
    boxByRef(b);
    cout << "box after boxByRef: " << b.v << endl;

    vector<int> v = {1, 2, 3};
    vecByValue(v);
    cout << "vector size after vecByValue: " << v.size() << endl;
    vecByRef(v);
    cout << "vector size after vecByRef: " << v.size() << endl;

    Box b1{1};
    Box b2 = b1;
    b2.v = 2;
    cout << "b1.v = " << b1.v << ", b2.v = " << b2.v << endl;

    vector<int> orig{1, 2, 3};
    vector<int> dup = orig;
    dup[0] = 100;
    cout << "orig[0] = " << orig[0] << ", dup[0] = " << dup[0] << endl;

    int* p = &n;
    *p = 50;
    cout << "n via pointer: " << n << endl;
    int& r = n;
    r = 60;
    cout << "n via reference: " << n << endl;
    return 0;
}
```
